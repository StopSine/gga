// Corrected function boundaries for Goemon's Great Adventure (USA).
//
// splat locates function starts heuristically and sometimes splits one routine
// into several. Declaring the true size here fixes it at the source, so both
// the disassembly and the linked ELF carry correct boundaries.
//
// Found by scanning for branches that escape their enclosing function:
// 11 escaping branches across 632 resident functions, forming 3 groups.

// Split three ways (func_80001088_1C88, func_8000121C_1E1C). Both tails branch
// back to .L80001190_1D90; the routine's only `jr $ra` is at 0x8000128C.
func_80000F10_1B10 = 0x80000F10; // size:0x384

// Split three ways (func_8000179C_239C, func_800017CC_23CC).
func_800013F0_1FF0 = 0x800013F0; // size:0x484

// Split two ways; the tail at func_80010294_10E94 branches back to 0x8001024C.
func_8000FD50_10950 = 0x8000FD50; // size:0x62C
