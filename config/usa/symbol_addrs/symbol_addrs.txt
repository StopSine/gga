// Corrected function boundaries for Goemon's Great Adventure (USA).
//
// splat locates function starts heuristically and sometimes splits one routine
// into several. Declaring the true size here fixes it at the source, so both
// the disassembly and the linked ELF carry correct boundaries.
//
// Found by scanning for branches that escape their enclosing function:
// 11 escaping branches across 632 resident functions, forming 3 groups.

// Split three ways (func_80001088_1C88, func_8000121C_1E1C). Both tails branch
// back to .L80001190_1D90; the routine's only `jr $ra` is at 0x8000128C.
func_80000F10_1B10 = 0x80000F10; // size:0x384

// Split three ways (func_8000179C_239C, func_800017CC_23CC).
func_800013F0_1FF0 = 0x800013F0; // size:0x484

// Split two ways; the tail at func_80010294_10E94 branches back to 0x8001024C.
func_8000FD50_10950 = 0x8000FD50; // size:0x62C

// Split four ways (func_80001FB4_2BB4, func_80001FC8_2BC8, func_80001FD8_2BD8).
// The escaping-branch scan above cannot find this one: the pieces are
// straight-line code that falls through, so no branch ever leaves the declared
// range. Found instead by the stack imbalance it causes -- the prologue
// `addiu $sp, $sp, -0x738` is in the first piece and the matching restore is in
// the last, so the recompiled first piece returned 0x738 below the stack it was
// called with. 0x20 + 0x14 + 0x10 + 0x88 = 0xCC, ending at the `jr $ra` at
// 0x80002058.
func_80001F94_2B94 = 0x80001F94; // size:0xCC

// A switch whose eleven cases splat made into separate functions, because each
// case is four instructions ending in `jr $ra`. The jump table at 0x0800C58C
// then pointed entirely outside the 0x3C the function was given, so N64Recomp
// could not size it and refused to recompile the function at all. Widening it
// to cover the cases (0x08001794..0x08001884) makes it an ordinary switch.
func_08001794_660634 = 0x08001794; // size:0xF0

// NOTE: a name is only safe to apply if the runtime implements
// <name>_recomp, or if N64Recomp will not rename it at all. Both
// reimplemented_funcs AND ignored_funcs cause the _recomp rename, and a
// renamed function the runtime does not supply is an undefined symbol at
// link time. 41 byte-matched names were withdrawn for that reason; left
// anonymous they are simply recompiled as ordinary code.

// Recovered once tools/gen_runtime_impls.py replaced the hand-maintained list
// of runtime-implemented functions. That list had drifted and was missing 25
// names, so valid matches were being rejected as unimplemented.
//
// osYieldThread and osCreateMesgQueue matter most here: the runtime already
// intercepts osSendMesg/osRecvMesg and the thread scheduler, so leaving the
// yield and the queue constructor recompiled let the game manage thread and
// queue state the scheduler never saw, and every game thread ended up parked
// in wait_for_resumed with nothing left to resume it.
//
// osMapTLB and osUnmapTLBAll were stubbed by address until now; the runtime
// fork implements both, so they are intercepted instead.

// Raw SI DMA, identified structurally rather than by byte match: GGA's copy
// only matches MNSG's at ~28%, but it is the same 0xAC bytes and touches
// exactly SI_DRAM_ADDR / PIF_ADDR_RD64B / PIF_ADDR_WR64B / SI_STATUS. GGA's
// controller layer calls it directly instead of going through osCont*, and
// the recompiled body would fault on the SI registers, so it is intercepted
// by the joybus implementation added to the runtime's cont.cpp.
__osSiRawStartDma = 0x80012C30;

// The VI framebuffer getters. osViSwapBuffer is intercepted, so the runtime
// owns which buffer is current, but these two were left recompiled and read
// the game's own __osViCurr/__osViNext globals at 0x8003BBF0/0x8003BBF4 --
// which the intercepted swap never updates. The game therefore drew using a
// stale framebuffer pointer, which showed up as split and offset logos.
//
// The byte match is exact but ambiguous between the two, since they differ
// only by the struct offset they load and the matcher masks immediates.
// Disambiguated by which global each one touches: in both games the pair is
// adjacent with Curr in the lower word, matching MNSG's 0x8007A300/0x8007A304.
osViGetCurrentFramebuffer = 0x800209F0;
osViGetNextFramebuffer = 0x80012840;

// The controller query path is deliberately NOT intercepted.
//
// GGA's own osContStartReadData is unnamed recompiled code, and libultra builds
// its poll block from state that osContInit sets up (__osMaxControllers and the
// PIF ram template). Intercepting osContInit replaced the function without
// establishing that state, so the read routine emitted an empty command list
// and the game never polled buttons -- it queried the controller once and then
// sent nothing. Letting the game's own init run means the whole stack is
// consistent, talking to the joybus implementation in the runtime.

//
// Names for routines whose behaviour has been established, so the disassembly
// carries the knowledge rather than it living only in commit messages. These
// reach a build through the ELF and --dump-context, not through asm/, so adding
// one changes nothing until the symbols are regenerated.
//
// When they are regenerated, anything in patches/ that names a function by its
// old func_<vram>_<rom> form stops resolving. gga.patches.toml sets
// strict_patch_mode, so that fails loudly rather than silently. Symbols the
// patches currently reference are therefore left unnamed and listed at the end.
//
// CAVEAT: splat 0.36.3 applies only some of these, exiting 0 and reporting
// nothing for the rest. Applied: objpool_free_all, objnode_create_child,
// objnode_attach_child, actor_init_from_spawn, scene_apply_scissor,
// flag_entry_set_bit1, gfx_context, and of the path and actor names added
// later, path_local_to_world, actor_get_world_position, player_slot_occupied,
// player_get_position and player_slots. Ignored: objpool_register,
// objpool_alloc, scenegraph_draw_node, scene_emit_camera_and_scissor,
// scene_render, scene_register_render, gfx_viewport, fade_display_list,
// path_eval_point and path_eval_tangent.
//
// Verify against the generated dumps rather than asm/. player_slots applied
// without producing any dlabel in asm/, so an asm/ grep called it dropped
// when it had in fact renamed and was about to break the patch build.
//
// It depends on the address rather than the name -- declaring 0x800D4A58 under
// a different name is ignored too -- and is not explained by duplicate
// declarations, the size overrides above, the type attribute,
// gga.undefined_syms.ld, or a stale cache. gfx_context is the one data symbol
// that took, and it is the one splat also emits into undefined_syms_auto.ld as
// a cross-section reference, which may be the distinguishing factor.
//
// The declarations are kept because they document what the routines are either
// way, and applying more of them later costs nothing.
//
// HAZARD when regenerating: the size:0xF0 override for func_08001794_660634 is
// at 0x08001794, an overlay address, and every overlay links at 0x08000000, so
// a bare address there is ambiguous across all 311 of them. splat resolved it
// on the run that produced the committed symbols and did not on a later one,
// giving that function 0x3C instead, which leaves its jump table unsizable and
// makes N64Recomp refuse to recompile it:
//
//   Failed to determine size of jump table at 0x0800C58C
//
// The generated gga.elf.syms.toml was corrected by hand for that one entry.
// Check it after any regeneration; the other four overrides, all at main-segment
// addresses, applied correctly.

// The path system, which is how GGA stores world positions. An actor does not
// hold a world x/z: it holds a distance along a path at +0x30, a lateral offset
// from that path at +0x38, and the path's id at +0x21. This is what makes the
// game a 2.5D side-scroller -- movement is along a spline, not in a plane.
//
// path_local_to_world(path_id, along, lateral, &out_x, &out_z) resolves one.
// The path table is at D_801094C0_5C46E0 + 0x20, indexed by id, 0xC ids; each
// entry has a live flag at +0xE, its curve at +0xC and its total length at
// +0x18. It passes (along, lateral) straight through when the id is 0, the
// table pointer is null, the id is out of range or the entry is not live;
// otherwise it clamps along to [0, length], evaluates the curve at
// along / length, and offsets the result perpendicular to the tangent there by
// lateral. Returns the clamped along value.
path_local_to_world = 0x800E3158; // type:func

// Evaluate a path's curve at a normalised parameter: the point, and the tangent
// at that point. Both take (out_vec3, curve, t), and both are wrappers that
// pick a segment and delegate -- to func_800E26DC_59D8FC and
// func_800E2B9C_59DDBC respectively. Named for the role their one caller,
// path_local_to_world, uses them in: the first result is a position, the second
// is normalised and rotated 90 degrees to form the lateral offset direction.
path_eval_point = 0x800E2900; // type:func
path_eval_tangent = 0x800E2DFC; // type:func

// Not named, because every overlay links at 0x08000000 and so an overlay
// address cannot be declared unambiguously (see the HAZARD note above). Each
// actor overlay carries its own byte-for-byte copy of a 59-line routine that
// returns the distance from the actor to player n via actor_get_world_position
// and player_get_position, plus a loop over the four slots returning the
// nearest. An actor wakes when that distance falls under a radius held as an
// immediate in the overlay's own code: 100.0 for the hellbat (file_172),
// file_169 and file_178, 130.0 for file_173, 70.0 for the nuppe (file_174),
// 80.0 for file_175 and the monoeye (file_176). patches/actor_range.c scales
// the reported distance so those radii track the aspect ratio.
